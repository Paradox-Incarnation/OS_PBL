cmake_minimum_required(VERSION 3.16)
project(SecureOS LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
add_library(secureos_core src/auth.cpp src/access_control.cpp src/audit_log.cpp src/file_manager.cpp src/encryption.cpp)
target_include_directories(secureos_core PUBLIC include)
target_link_libraries(secureos_core PUBLIC OpenSSL::Crypto)
if(MSVC)
  target_compile_options(secureos_core PRIVATE /W4)
else()
  target_compile_options(secureos_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(secureos src/main.cpp src/ui.cpp)
target_link_libraries(secureos PRIVATE secureos_core)
add_executable(secureos_tests tests/test_secureos.cpp)
target_link_libraries(secureos_tests PRIVATE secureos_core)
enable_testing()
add_test(NAME secureos_tests COMMAND secureos_tests)
